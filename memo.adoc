    where에 사용되는 칼럼에 index 적용.
    Cursor 기반 페이징
    Cursor 기반 페이징 방식은 조회 시작 부분을 인덱스로 빠르게 찾아 매번 첫 페이지만 읽도록 하는 방식이다.
    아무리 페이지가 뒤로 가더라도 처음 페이지를 읽은 것과 동일한 성능을 가지게 되므로, Offset 방식보다 훨씬 성능이 개선된다.
    그러나, 이 방식은 원래 페이지 번호가 없고, 다음 페이지를 호출할 수 있는 키워드 등을 제공하는 페이징 기능이다. 아래와 같은 시스템 (흔히 SNS 게시물 조회 API에서 많이 볼 수 있다) 에서 사용한다.

-- Users 테이블
CREATE TABLE Users (
                       id INT AUTO_INCREMENT PRIMARY KEY,
                       username VARCHAR(255) NOT NULL,
                       password VARCHAR(255) NOT NULL,
                       email VARCHAR(255) NOT NULL,
                       phone VARCHAR(20),
                       role ENUM('CUSTOMER', 'SELLER', 'ADMIN') NOT NULL,
                       created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Products 테이블
CREATE TABLE Products (
                          id INT AUTO_INCREMENT PRIMARY KEY,
                          name VARCHAR(255) NOT NULL,
                          description TEXT,
                          price DECIMAL(10, 2) NOT NULL,
                          stock INT NOT NULL,
                          category_id INT,
                          created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                          updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
                          FOREIGN KEY (category_id) REFERENCES Categories(id)
);

-- Categories 테이블
CREATE TABLE Categories (
                            id INT AUTO_INCREMENT PRIMARY KEY,
                            name VARCHAR(255) NOT NULL,
                            description TEXT
);

-- Orders 테이블
CREATE TABLE Orders (
                        id INT AUTO_INCREMENT PRIMARY KEY,
                        user_id INT,
                        order_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                        total_amount DECIMAL(10, 2) NOT NULL,
                        status ENUM('PENDING', 'COMPLETED', 'SHIPPING', 'DELIVERED', 'CANCELED') NOT NULL,
                        payment_method VARCHAR(50),
                        FOREIGN KEY (user_id) REFERENCES Users(id)
);

-- Order_Items 테이블
CREATE TABLE Order_Items (
                             id INT AUTO_INCREMENT PRIMARY KEY,
                             order_id INT,
                             product_id INT,
                             quantity INT NOT NULL,
                             price DECIMAL(10, 2) NOT NULL,
                             total_price DECIMAL(10, 2) NOT NULL,
                             FOREIGN KEY (order_id) REFERENCES Orders(id),
                             FOREIGN KEY (product_id) REFERENCES Products(id)
);

-- Shopping_Carts 테이블
CREATE TABLE Shopping_Carts (
                                id INT AUTO_INCREMENT PRIMARY KEY,
                                user_id INT,
                                product_id INT,
                                quantity INT NOT NULL,
                                added_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                                FOREIGN KEY (user_id) REFERENCES Users(id),
                                FOREIGN KEY (product_id) REFERENCES Products(id)
);

-- Payments 테이블
CREATE TABLE Payments (
                          id INT AUTO_INCREMENT PRIMARY KEY,
                          order_id INT,
                          payment_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                          amount DECIMAL(10, 2) NOT NULL,
                          payment_method VARCHAR(50),
                          status ENUM('SUCCESS', 'FAILED', 'PENDING') NOT NULL,
                          FOREIGN KEY (order_id) REFERENCES Orders(id)
);

-- Addresses 테이블
CREATE TABLE Addresses (
                           id INT AUTO_INCREMENT PRIMARY KEY,
                           user_id INT,
                           recipient_name VARCHAR(255) NOT NULL,
                           address_line1 VARCHAR(255) NOT NULL,
                           address_line2 VARCHAR(255),
                           city VARCHAR(100) NOT NULL,
                           postal_code VARCHAR(20) NOT NULL,
                           phone VARCHAR(20) NOT NULL,
                           FOREIGN KEY (user_id) REFERENCES Users(id)
);

-- Reviews 테이블
CREATE TABLE Reviews (
                         id INT AUTO_INCREMENT PRIMARY KEY,
                         user_id INT,
                         product_id INT,
                         rating INT CHECK (rating BETWEEN 1 AND 5),
                         comment TEXT,
                         created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                         FOREIGN KEY (user_id) REFERENCES Users(id),
                         FOREIGN KEY (product_id) REFERENCES Products(id)
);

-- Wishlists 테이블
CREATE TABLE Wishlists (
                           id INT AUTO_INCREMENT PRIMARY KEY,
                           user_id INT,
                           product_id INT,
                           added_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                           FOREIGN KEY (user_id) REFERENCES Users(id),
                           FOREIGN KEY (product_id) REFERENCES Products(id)
);

-- Product_Images 테이블
CREATE TABLE Product_Images (
                                id INT AUTO_INCREMENT PRIMARY KEY,
                                product_id INT,
                                image_url VARCHAR(255) NOT NULL,
                                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                                FOREIGN KEY (product_id) REFERENCES Products(id)
);

-- Coupons 테이블
CREATE TABLE Coupons (
                         id INT AUTO_INCREMENT PRIMARY KEY,
                         code VARCHAR(50) NOT NULL,
                         discount_amount DECIMAL(10, 2) NOT NULL,
                         valid_from DATE NOT NULL,
                         valid_to DATE NOT NULL,
                         user_id INT,
                         FOREIGN KEY (user_id) REFERENCES Users(id) ON DELETE SET NULL
);


-- Merchants 테이블 (가맹점 정보)
CREATE TABLE Merchants (
                           id INT AUTO_INCREMENT PRIMARY KEY,
                           name VARCHAR(255) NOT NULL,
                           email VARCHAR(255) NOT NULL,
                           phone VARCHAR(20),
                           address VARCHAR(255),
                           created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                           status ENUM('ACTIVE', 'INACTIVE') NOT NULL DEFAULT 'ACTIVE'
);

-- Merchant_Products 테이블 (가맹점과 상품 간의 관계)
CREATE TABLE Merchant_Products (
                                   merchant_id INT,
                                   product_id INT,
                                   PRIMARY KEY (merchant_id, product_id),
                                   FOREIGN KEY (merchant_id) REFERENCES Merchants(id) ON DELETE CASCADE,
                                   FOREIGN KEY (product_id) REFERENCES Products(id) ON DELETE CASCADE
);

-- Settlements 테이블 (정산 내역)
CREATE TABLE Settlements (
                             id INT AUTO_INCREMENT PRIMARY KEY,
                             merchant_id INT,
                             total_sales DECIMAL(15, 2) NOT NULL,
                             settlement_amount DECIMAL(15, 2) NOT NULL,
                             settlement_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                             FOREIGN KEY (merchant_id) REFERENCES Merchants(id)
);
